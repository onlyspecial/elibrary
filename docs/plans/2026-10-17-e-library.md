# 电子图书馆服务 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 `source/` 下落地可本地运行的电子图书馆 Web 服务（实体借还 + 电子书 PDF 在线预览 + 角色权限）。

**Architecture:** Spring Boot 单实例单体；Controller → Service → MyBatis Mapper；Thymeleaf SSR；Spring Security Session + BCrypt；PDF 本地磁盘；借阅/归还用 Redis `SET NX` + TTL 分布式锁。

**Tech Stack:** Java 8、Spring Boot 2.7.x、Spring MVC、Thymeleaf、MyBatis、MySQL 5.7、Spring Security、Spring Data Redis、Maven、JUnit 5 + Mockito + Spring Boot Test（`javax.*`，非 `jakarta.*`）

**Git 约定：** 工作区尚未建仓；本计划中的 Commit 步骤一律 **跳过**，直至用户明确要求 `git init` / commit。

## 开放细节锁定（相对 design §14）

| 项 | 决策 |
|----|------|
| 借阅入口 | 读者对**书目**一键借阅；服务端选一册 `AVAILABLE` 且未删除的复本 |
| Redis 锁 | `StringRedisTemplate` + `SET key value NX EX ttl`（TTL=10s）；锁 value 用 UUID，释放时校验 value（Lua 或 get+del 谨慎释放）；**不引入 Redisson** |
| PDF 预览 | 独立预览端点 `Content-Disposition: inline`；详情页用 `<iframe>` 指向该端点；**无下载按钮、无 attachment 接口** |
| 存储目录 | 启动时若 `library.pdf.storage-dir` 不存在则 `Files.createDirectories` |
| 基包名 | `com.elibrary` |
| Spring Boot | **2.7.18**，**Java 8**；Thymeleaf Security 用 `thymeleaf-extras-springsecurity5` |
| MySQL | **5.7**；DDL 使用 `ENGINE=InnoDB DEFAULT CHARSET=utf8mb4` |
| 配置键 | Boot 2：`spring.redis.*`（非 Boot 3 的 `spring.data.redis`） |
| 校验包 | `javax.validation`（非 `jakarta.validation`） |
| 分页 | 简单 `page`/`size` 查询参数 + LIMIT/OFFSET；默认 size=10 |
| 锁失败文案 | 「借阅繁忙，请稍后重试」 |
| 无可用复本 | 「暂无可用复本」 |

import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.ArgumentMatchers;
import org.mockito.InjectMocks;
import org.mockito.Mock;
import org.mockito.junit.jupiter.MockitoExtension;
import org.springframework.data.redis.core.StringRedisTemplate;
import org.springframework.data.redis.core.ValueOperations;
import org.springframework.data.redis.core.script.RedisScript;

import java.time.Duration;
import java.util.Collections;
import java.util.Optional;

import static org.assertj.core.api.Assertions.assertThat;
import static org.mockito.ArgumentMatchers.*;
import static org.mockito.Mockito.*;

@ExtendWith(MockitoExtension.class)
class RedisDistributedLockTest {

    @Mock StringRedisTemplate redisTemplate;
    @Mock ValueOperations<String, String> valueOps;
    @InjectMocks RedisDistributedLock lock;

    @Test
    @DisplayName("抢锁-当key不存在时-应返回token")
    void tryLock_whenKeyAbsent_shouldReturnToken() {
        // Arrange
        when(redisTemplate.opsForValue()).thenReturn(valueOps);
        when(valueOps.setIfAbsent(eq("borrow:copy:1"), anyString(), eq(Duration.ofSeconds(10))))
                .thenReturn(true);

        // Act
        Optional<String> token = lock.tryLock("borrow:copy:1", Duration.ofSeconds(10));

        // Assert
        assertThat(token).isPresent();
        assertThat(token.get()).isNotBlank();
    }

    @Test
    @DisplayName("抢锁-当key已存在时-应返回empty")
    void tryLock_whenKeyPresent_shouldReturnEmpty() {
        when(redisTemplate.opsForValue()).thenReturn(valueOps);
        when(valueOps.setIfAbsent(anyString(), anyString(), any(Duration.class))).thenReturn(false);

        Optional<String> token = lock.tryLock("borrow:copy:1", Duration.ofSeconds(10));

        assertThat(token).isEmpty();
    }

    @Test
    @DisplayName("释放锁-当token匹配时-应删除key")
    void unlock_whenTokenMatches_shouldDeleteKey() {
        when(redisTemplate.execute(any(RedisScript.class), eq(Collections.singletonList("borrow:copy:1")), eq("tok")))
                .thenReturn(1L);

        lock.unlock("borrow:copy:1", "tok");

        verify(redisTemplate).execute(any(RedisScript.class), eq(Collections.singletonList("borrow:copy:1")), eq("tok"));
    }
}

```java
package com.elibrary.common.lock;

import org.springframework.data.redis.core.StringRedisTemplate;
import org.springframework.data.redis.core.script.DefaultRedisScript;
import org.springframework.stereotype.Component;

import java.time.Duration;
import java.util.Collections;
import java.util.Optional;
import java.util.UUID;

@Component
public class RedisDistributedLock {

    private static final String UNLOCK_LUA =
            "if redis.call('get', KEYS[1]) == ARGV[1] then return redis.call('del', KEYS[1]) else return 0 end";

    private final StringRedisTemplate redisTemplate;
    private final DefaultRedisScript<Long> unlockScript;

    public RedisDistributedLock(StringRedisTemplate redisTemplate) {
        this.redisTemplate = redisTemplate;
        this.unlockScript = new DefaultRedisScript<>();
        this.unlockScript.setScriptText(UNLOCK_LUA);
        this.unlockScript.setResultType(Long.class);
    }

    public Optional<String> tryLock(String key, Duration ttl) {
        String token = UUID.randomUUID().toString();
        Boolean ok = redisTemplate.opsForValue().setIfAbsent(key, token, ttl);
        if (Boolean.TRUE.equals(ok)) {
            return Optional.of(token);
        }
        return Optional.empty();
    }

    public void unlock(String key, String token) {
        redisTemplate.execute(unlockScript, Collections.singletonList(key), token);
    }
}

package com.elibrary.user;

import com.elibrary.common.BusinessException;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class UserService {
    private final UserMapper userMapper;
    private final PasswordEncoder passwordEncoder;

    public UserService(UserMapper userMapper, PasswordEncoder passwordEncoder) {
        this.userMapper = userMapper;
        this.passwordEncoder = passwordEncoder;
    }

    @Transactional
    public User register(String username, String rawPassword) {
        if (userMapper.findByUsername(username) != null) {
            throw new BusinessException("用户名已存在");
        }
        User user = new User();
        user.setUsername(username);
        user.setPasswordHash(passwordEncoder.encode(rawPassword));
        user.setRole(Role.READER);
        userMapper.insert(user);
        return user;
    }

    public User requireById(Long id) {
        User user = userMapper.findById(id);
        if (user == null) {
            throw new BusinessException("用户不存在");
        }
        return user;
    }

    @Transactional
    public void grantAdmin(Long userId) {
        User user = requireById(userId);
        if (user.getRole() == Role.SUPER_ADMIN) {
            throw new BusinessException("不能变更超级管理员角色");
        }
        userMapper.updateRole(userId, Role.ADMIN);
        user.setRole(Role.ADMIN);
    }

    @Transactional
    public void revokeAdmin(Long userId) {
        User user = requireById(userId);
        if (user.getRole() == Role.SUPER_ADMIN) {
            throw new BusinessException("不能撤销超级管理员");
        }
        if (user.getRole() != Role.ADMIN) {
            throw new BusinessException("目标用户不是管理员");
        }
        userMapper.updateRole(userId, Role.READER);
        user.setRole(Role.READER);
    }

    public User findByUsername(String username) {
        return userMapper.findByUsername(username);
    }
}

        }
        BorrowRecord record = new BorrowRecord();
        record.setUserId(userId);
        record.setPhysicalCopyId(copy.getId());
        record.setBorrowedAt(LocalDateTime.now());
        borrowRecordMapper.insert(record);
        copyMapper.updateStatus(copy.getId(), CopyStatus.BORROWED);
    } finally {
        lock.unlock(key, token);
    }
}

@Transactional
public void returnByRecordId(Long userId, Long recordId) {
    BorrowRecord record = borrowRecordMapper.findById(recordId);
    if (record == null || record.getReturnedAt() != null) {
        throw new BusinessException("借阅记录不存在或已归还");
    }
    if (!record.getUserId().equals(userId)) {
        throw new BusinessException("只能归还自己的借阅");
    }
    String key = "borrow:copy:" + record.getPhysicalCopyId();
    String token = lock.tryLock(key, Duration.ofSeconds(10))
            .orElseThrow(() -> new BusinessException("归还繁忙，请稍后重试"));
    try {
        borrowRecordMapper.markReturned(recordId, LocalDateTime.now());
        copyMapper.updateStatus(record.getPhysicalCopyId(), CopyStatus.AVAILABLE);
    } finally {
        lock.unlock(key, token);
    }
}
```

注意：
- `findByIdForUpdate` 使用 `SELECT ... FOR UPDATE`（MySQL）；测试用 H2 时可改为普通 select 或测试 profile 关闭 FOR UPDATE——**单测以 Mockito 为主，不强制启动完整 DB**。
- Controller 从 `Authentication` 取 username，再查 `User.id`；或自定义 `UserDetails` 携带 id（推荐扩展 `LibraryUserPrincipal` 带 `userId`，避免每次查库——若做此扩展，在 Task 3 的 `LibraryUserDetailsService` 一并改，测试同步）。

**推荐：`LibraryUserPrincipal` 实现 `UserDetails`，字段 `id/username/passwordHash/role`。** 若 Task 3 已用标准 User，本 Task 开头先重构再写借阅。
